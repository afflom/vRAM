  name: Request, type_parameters: [], parameters: [],

definition{
  name: search, parameters: [request : Request], result: Bool,
  body: (match request
    | Request.explore incumbent nodeBudget => ((256 <? incumbent) && (0 <? nodeBudget))
    | Request.split extent remaining => ((16 <? extent) && (0 <? remaining))
    | Request.replace incumbent challenger exact => (exact && (challenger <? incumbent))
    | Request.factor classes axis => ((0 <? classes) && ((classes <? axis) && (classes <=? 16)))),
  executable: true, production: {effects: [], targets: [rust-core]}
}
