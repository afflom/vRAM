\title{Natural number}
\begin{semanticmodule}
definition{
  name: matchesCell,
  parameters: [right : Bool, leftByte : Nat, rightByte : Nat, observed : Nat],
  result: Bool,
  body: (if right then (observed =? rightByte) else (observed =? leftByte)),
  executable: true,
  production: {effects: [], targets: [rust-core]}
}

theorem{
  name: leftAdmission,
  parameters: [a : Nat, b : Nat, observed : Nat],
  statement: ((matchesCell false a b observed) = (observed =? a)),
  proof: reflexivity{}
}

theorem{
  name: rightAdmission,
  parameters: [a : Nat, b : Nat, observed : Nat],
  statement: ((matchesCell true a b observed) = (observed =? b)),
  proof: reflexivity{}
}
\end{semanticmodule}
