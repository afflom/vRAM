definition{
  name: inlineFill, parameters: [uniform : Bool], result: Bool,
  body: (match (Main::choose uniform 4096 4096 4096 4096)
    | Main::Representation.fill => true
    | Main::Representation.sparse => false
    | Main::Representation.palette => false
    | Main::Representation.raw => false
    | Main::Representation.selector => false
    | Main::Representation.corrected => false), executable: true
}
theorem{ name: uniformFill, parameters: [], statement: ((inlineFill true) = true), proof: reflexivity{} }
theorem{ name: denseLiteral, parameters: [], statement: ((inlineFill false) = false), proof: reflexivity{} }
definition{
  name: residentBytes, parameters: [directory : Nat, payload : Nat], result: Nat,
  body: (directory + payload), executable: true
}
definition{
  name: charged, parameters: [directory : Nat, literals : Nat], result: Nat,
  body: (residentBytes directory (prim multiply literals 4096 : Nat)), executable: true
}
theorem{
  name: breakEven, parameters: [directory : Nat, literalBytes : Nat, fillBytes : Nat],
  statement: ((Performance::disagree ((residentBytes directory literalBytes) <? (literalBytes + fillBytes))
    (directory <? fillBytes)) = false),
  proof: linear_arithmetic{definitions: [Performance::disagree, residentBytes]}, axioms: [Classical.choice, Quot.sound, propext]
}
definition{
  name: span, parameters: [op : Nat, aligned : Bool, completePages : Bool, inside : Bool], result: Nat,
  body: (LiteralBioPolicy::action op aligned completePages inside), executable: true
}

theorem{ name: writeSpan, parameters: [], statement: ((span 1 true true true) = 2), proof: reflexivity{} }

theorem{ name: unsupportedSpan, parameters: [], statement: ((span 2 true true true) = 0), proof: reflexivity{} }

  name: heldWord, parameters: [payload : Nat], result: Nat,
  body: (payload + (lockMask)), executable: true
}
theorem{
  name: fillBelowLock, parameters: [byte : Nat],
  statement: (((byte <? 256) && ((lockMask) <=? byte)) = false),
  proof: linear_arithmetic{definitions: [lockMask]}, axioms: [Classical.choice, Quot.sound, propext]
}
theorem{
  name: packedFillBelowFrame, parameters: [byte : Nat],
  statement: (((byte <? 256) && (4096 <=? (heldWord byte))) = false),

  name: packedLiteralAboveFill, parameters: [address : Nat],

theorem{
  name: unlockSlotPayload, parameters: [payload : Nat],
  statement: ((prim subtract (heldWord payload) (lockMask) : Nat) = payload),

  name: foldStep, parameters: [order : Nat, equalHalves : Bool], result: Nat,
