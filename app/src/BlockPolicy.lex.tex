  name: isReply, parameters: [phase : Nat], result: Bool,
  body: (((phase =? (ack)) || (phase =? (invalid))) || ((phase =? (unsupported)) || (phase =? (readError)))), executable: true
}
definition{
  name: beginRequest, parameters: [command : Nat, allowed : Bool, empty : Bool], result: Nat,
  body: (if (command =? 2) then (disconnect) else
    (if (command =? 3) then (if empty then (ack) else (invalid)) else
     (if allowed then
       (if (command =? 0) then (readHead) else (if (command =? 1) then (writePage) else
         (if (command =? 4) then (trimRange) else (unsupported))))
      else (if ((command =? 1) && (not empty)) then (drain) else (invalid))))), executable: true
}
definition{
  name: step, parameters: [phase : Nat, command : Nat, allowed : Bool, empty : Bool, success : Bool, last : Bool], result: Nat,
  body: (if (phase =? (header)) then (if success then (beginRequest command allowed empty) else (abortRequest)) else
    (if (not success) then (if (phase =? (readHead)) then (readError) else (abortRequest)) else
     (if (phase =? (readHead)) then (sendHead) else
      (if (phase =? (readTail)) then (sendTail) else
       (if ((phase =? (sendHead)) || (phase =? (sendTail))) then (if last then (done) else (readTail)) else
        (if (phase =? (writePage)) then (if last then (ack) else (writePage)) else
         (if (phase =? (trimRange)) then (ack) else
          (if (phase =? (drain)) then (if last then (invalid) else (drain)) else
           (if (isReply phase) then (done) else (abortRequest)))))))))),
  executable: true, production: {effects: [], targets: [rust-core]}
}
